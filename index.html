/** * Update Progress — Backend API Multi-Kategori (Google Apps Script) * ---------------------------------------------------------------------- * Setup: * 1. Spreadsheet ini punya tab per kategori, masing-masing formatnya sama: * IHLD | NO ORDER | NAMA PROYEK | DISTRICT | Progress Lapangan | Panjang Kabel (m) | Jenis Kabel * Nama tab: OLO, HEM, PT3 TA, PT3 FH, MBB, QE * (kalau ada kategori baru, tinggal tambah namanya di array KATEGORI_LIST di bawah) * * 2. Sheet "Data Deploy" (dulu Material_Master), header baris 1: * Kategori | NamaMaterial | Satuan | JenisMaterial * * 3. Sheet "Data Progress" (dulu Material_Input, biarkan kosong, akan otomatis terisi): * Timestamp | Kategori | IHLD | NoOrder | NamaProyek | Petugas | Material | JumlahActual | JumlahPlan | Satuan | FotoURL | Koordinat | Catatan * * 4. (Opsional) Buat folder Google Drive buat foto bukti, isi ID-nya ke DRIVE_FOLDER_ID. * * 5. Extensions > Apps Script di spreadsheet ini, paste seluruh file ini. * 6. Deploy > New deployment > Web app (Execute as: Me, Access: Anyone) * 7. Copy URL -> pakai sebagai API_URL di index_Material.html */ const KATEGORI_LIST = ['OLO', 'HEM', 'PT3 TA', 'PT3 FH', 'MBB', 'QE']; const SHEET_DATA_DEPLOY = 'Data Deploy'; const SHEET_DATA_PROGRESS = 'Data Progress'; const DRIVE_FOLDER_ID = ''; // isi Folder ID Google Drive kalau mau simpan foto function doGet(e) { const params = e.parameter || {}; try { let result; if (params.action === 'search_order') { result = { orders: searchOrders(params.kategori, params.q || '') }; } else if (params.action === 'get_materials') { result = { materials: getDataDeploy(params.kategori) }; } else if (params.action === 'get_kategori_list') { result = { kategoriList: KATEGORI_LIST }; } else if (params.action === 'get_order_detail') { result = getOrderDetail(params.kategori, params.orderId); } else { result = { kategoriList: KATEGORI_LIST }; } return jsonOutput(result); } catch (err) { return jsonOutput({ error: err.message }); } } // Dipakai mode "Cek Progress JT" (read-only): gabungkan data dasar order // (dari tab kategori) + histori update terakhir tiap material (dari Data Progress) function getOrderDetail(kategori, orderId) { const order = getOrderById(kategori, orderId); const progress = getProgressRows(kategori, orderId); return { order: order, progress: progress }; } function getOrderById(kategori, orderId) { if (!kategori || KATEGORI_LIST.indexOf(kategori) === -1) { throw new Error('Kategori tidak valid: ' + kategori); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(kategori); if (!sheet) throw new Error('Sheet kategori "' + kategori + '" tidak ditemukan'); const values = sheet.getDataRange().getValues(); values.shift(); const row = values.find(r => String(r[0]) === String(orderId)); if (!row) return null; return { orderId: String(row[0]), noOrder: row[1], namaProyek: row[2], district: row[3], progressLapangan: row[4], panjangKabelPlan: row[5], jenisKabel: row[6] }; } // Kolom Data Progress: Timestamp | Kategori | IHLD | NoOrder | NamaProyek | Petugas | Material | JumlahActual | JumlahPlan | Satuan | FotoURL | Koordinat | Catatan function getProgressRows(kategori, orderId) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_DATA_PROGRESS); const values = sheet.getDataRange().getValues(); values.shift(); return values .filter(r => String(r[1]) === kategori && String(r[2]) === String(orderId)) .map(r => ({ timestamp: r[0] instanceof Date ? r[0].toISOString() : String(r[0]), petugas: r[5], material: r[6], jumlahActual: r[7], jumlahPlan: r[8], satuan: r[9], fotoUrl: r[10], koordinat: r[11], catatan: r[12] })); } function doPost(e) { try { const body = JSON.parse(e.postData.contents); const { kategori, orderId, noOrder, namaProyek, petugas, materials, koordinat, catatan, fotoBase64 } = body; if (!kategori) throw new Error('Kategori wajib diisi'); if (!orderId) throw new Error('Order ID wajib diisi'); if (!materials || !materials.length) throw new Error('Minimal 1 material harus diisi'); let fotoUrl = ''; if (fotoBase64 && DRIVE_FOLDER_ID) { fotoUrl = saveFotoToDrive(fotoBase64, orderId); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_DATA_PROGRESS); const timestamp = new Date(); // Hapus baris lama yang match Kategori + IHLD + Material yang sama (replace, bukan tambah riwayat) removeExistingProgressRows(sheet, kategori, orderId, materials.map(m => m.nama)); materials.forEach(m => { sheet.appendRow([ timestamp, kategori, orderId, // IHLD noOrder || '', namaProyek || '', petugas || '', m.nama, m.jumlahActual, m.jumlahPlan || '', m.satuan || '', fotoUrl, koordinat || '', catatan || '' ]); }); return jsonOutput({ success: true, itemsSaved: materials.length }); } catch (err) { return jsonOutput({ success: false, error: err.message }); } } // Hapus baris di Data Progress yang punya Kategori + IHLD + nama Material sama // (dipanggil sebelum insert baris baru, supaya "replace" bukan "tambah riwayat") // Kolom Data Progress: Timestamp | Kategori | IHLD | NoOrder | NamaProyek | Petugas | Material | JumlahActual | JumlahPlan | Satuan | FotoURL | Koordinat | Catatan function removeExistingProgressRows(sheet, kategori, orderId, materialNames) { const data = sheet.getDataRange().getValues(); const rowsToDelete = []; for (let i = data.length - 1; i >= 1; i--) { // dari bawah, skip header (baris 0) const rowKategori = String(data[i][1] || ''); const rowOrderId = String(data[i][2] || ''); // IHLD const rowMaterial = String(data[i][6] || ''); // Material if (rowKategori === kategori && rowOrderId === orderId && materialNames.includes(rowMaterial)) { rowsToDelete.push(i + 1); } } rowsToDelete.forEach(rowNum => sheet.deleteRow(rowNum)); } // Cari order di tab sesuai kategori yang dipilih. // Kolom tab kategori: IHLD | NO ORDER | NAMA PROYEK | DISTRICT | Progress Lapangan | Panjang Kabel (m) | Jenis Kabel function searchOrders(kategori, query) { if (!kategori || KATEGORI_LIST.indexOf(kategori) === -1) { throw new Error('Kategori tidak valid: ' + kategori); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(kategori); if (!sheet) throw new Error('Sheet kategori "' + kategori + '" tidak ditemukan'); const values = sheet.getDataRange().getValues(); values.shift(); // buang header const q = query.trim().toLowerCase(); let rows = values.filter(r => r[0]); // ada IHLD if (q.length >= 3) { rows = rows.filter(r => String(r[0]).toLowerCase().includes(q) || // IHLD String(r[1]).toLowerCase().includes(q) || // NO ORDER String(r[2]).toLowerCase().includes(q) // NAMA PROYEK ); } else if (q.length > 0) { return []; } else { rows = rows.slice(-20); } return rows.map(r => ({ orderId: String(r[0]), noOrder: r[1], namaProyek: r[2], district: r[3], progressLapangan: r[4], panjangKabelPlan: r[5], jenisKabel: r[6] })); } // Ambil daftar material dari Data Deploy, difilter sesuai kategori // Kolom Data Deploy: Kategori | NamaMaterial | Satuan | JenisMaterial function getDataDeploy(kategori) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_DATA_DEPLOY); const values = sheet.getDataRange().getValues(); values.shift(); let rows = values.filter(r => r[1]); // ada NamaMaterial if (kategori) { rows = rows.filter(r => String(r[0]).trim() === kategori); } return rows.map(r => ({ kategori: r[0], nama: r[1], satuan: r[2], jenis: r[3] })); } function saveFotoToDrive(base64Data, orderId) { const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const matches = base64Data.match(/^data:(image\/\w+);base64,(.+)$/); if (!matches) throw new Error('Format foto tidak valid'); const mimeType = matches[1]; const bytes = Utilities.base64Decode(matches[2]); const ext = mimeType.split('/')[1] || 'jpg'; const blob = Utilities.newBlob(bytes, mimeType, `${orderId}_${Date.now()}.${ext}`); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } function jsonOutput(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
